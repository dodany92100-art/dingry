  id: number;
  name: string;
  variant: string;
  price: number;
  image: string;
  /** Dos (CRM) — prioritaire sur les paires figées hoodies/sweats si renseigné. */
  imageBack?: string;
  description: string;
  fabrication?: string;
  limitedEdition?: string;
  materials?: string;
  sizeGuide?: string;
  printPlacement?: ProductPrintPlacement;
  category: ShopCategory | string;
  sku?: string;
  sizes?: string[];
  colorDeclinations?: string[];
  /** Options parent (couleur…) — visuels optionnels, SKU combiné au checkout. */
  options?: import("@/lib/catalog/product-options").ShopProductOption[];
  /** 0 = rupture (API `inStock: false`), absent = fallback statique en stock. */
  stock?: number;
  isPinned?: boolean;
  displayOrder?: number;
  /** ISO — tri antichrono boutique si pas d’ordre manuel. */
  adminCreatedAt?: string;
};

export function isShopProductInStock(product: ShopProduct): boolean {
  if (product.stock === undefined) return true;
  return product.stock > 0;
}

function storyFieldsFromCatalog(product: CatalogProduct): Pick<
  ShopProduct,
  "fabrication" | "limitedEdition" | "materials" | "sizeGuide" | "printPlacement"
> {
  return {
    ...(product.fabrication?.trim() ? { fabrication: product.fabrication.trim() } : {}),
    ...(product.limitedEdition?.trim() ? { limitedEdition: product.limitedEdition.trim() } : {}),
    ...(product.materials?.trim() ? { materials: product.materials.trim() } : {}),
    ...(product.sizeGuide?.trim() ? { sizeGuide: product.sizeGuide.trim() } : {}),
    ...(product.printPlacement ? { printPlacement: product.printPlacement } : {}),
  };
}

function shopOptionsForCatalog(product: CatalogProduct) {
  if (product.options && product.options.length > 0) {
    return shopOptionsFromProductOptions(product.options);
  }
  if (product.colorDeclinations && product.colorDeclinations.length > 0) {
    return shopOptionsFromColorDeclinations(product.colorDeclinations, product.variant, {
      parentSku: product.sku,
      productName: product.name,
    });
  }
  return undefined;
}

export function catalogToShopProduct(product: CatalogProduct): ShopProduct {
  const options = shopOptionsForCatalog(product);
  return {
    id: product.id,
    name: product.name,
    variant: product.variant,
    price: product.price,
    image: product.image,
    ...(product.imageBack?.trim() ? { imageBack: product.imageBack.trim() } : {}),
    description: product.description,
    ...storyFieldsFromCatalog(product),
    category: resolveShopCategory(product.id, product.category),
    sku: product.sku,
    sizes: product.sizes,
    stock: product.stock,
    ...(product.colorDeclinations?.length ? { colorDeclinations: [...product.colorDeclinations] } : {}),
    ...(options?.length ? { options } : {}),
    ...(product.isPinned ? { isPinned: true } : {}),
    ...(product.displayOrder !== undefined && product.displayOrder > 0
      ? { displayOrder: product.displayOrder }
      : {}),
    ...(product.adminCreatedAt ? { adminCreatedAt: product.adminCreatedAt } : {}),
  };
}

function resolveLiveOverlay(
  productId: number,
  liveById: Map<number, CatalogProduct>,
): { overlay: CatalogProduct; inheritedFromRecto: boolean } | null {
  const direct = liveById.get(productId);
  if (direct) return { overlay: direct, inheritedFromRecto: false };

  if (!isVersoCatalogId(productId)) return null;
  const rectoId = resolveRectoCatalogId(productId);
  if (rectoId === productId) return null;
  const fromRecto = liveById.get(rectoId);
  if (!fromRecto) return null;
  return { overlay: fromRecto, inheritedFromRecto: true };
}

function mergeStoryOverlay(
  base: ShopProduct,
  overlay: CatalogProduct,
): Pick<ShopProduct, "fabrication" | "limitedEdition" | "materials" | "sizeGuide" | "printPlacement"> {
  const pick = (key: "fabrication" | "limitedEdition" | "materials" | "sizeGuide") => {
    const fromOverlay = overlay[key]?.trim();
    if (fromOverlay) return fromOverlay;
    return base[key]?.trim() || undefined;
  };
  const printPlacement = overlay.printPlacement || base.printPlacement;
  return {
    ...(pick("fabrication") ? { fabrication: pick("fabrication") } : {}),
    ...(pick("limitedEdition") ? { limitedEdition: pick("limitedEdition") } : {}),
    ...(pick("materials") ? { materials: pick("materials") } : {}),
    ...(pick("sizeGuide") ? { sizeGuide: pick("sizeGuide") } : {}),
    ...(printPlacement ? { printPlacement } : {}),
  };
}

function applyCatalogOverlay(
  product: ShopProduct,
  overlay: CatalogProduct,
  inheritedFromRecto: boolean,
): ShopProduct {
  const liveImage = overlay.image.trim();
  const sku =
    inheritedFromRecto && product.id !== overlay.id
      ? versoSkuFromRecto(overlay.sku)
      : overlay.sku;
  const overlayBack = overlay.imageBack?.trim();
  const crmImage =
    liveImage && !isDraftPlaceholderImage(liveImage)
      ? liveImage
      : (CATALOG_CANONICAL_IMAGE_BY_ID[product.id] ?? product.image);
  return {
    ...product,
    name: overlay.name,
    variant: inheritedFromRecto
      ? product.variant || overlay.variant
      : overlay.variant || product.variant,
    price: overlay.price,
    image: inheritedFromRecto ? product.image : crmImage,
    ...(inheritedFromRecto
      ? {}
      : overlayBack && !isDraftPlaceholderImage(overlayBack)
        ? { imageBack: overlayBack }
        : { imageBack: undefined }),
    description: overlay.description?.trim() || product.description,
    ...mergeStoryOverlay(product, overlay),
    category: resolveShopCategory(product.id, overlay.category || product.category),
    sku,
    sizes: overlay.sizes,
    stock: overlay.stock,
    ...(overlay.colorDeclinations?.length
      ? { colorDeclinations: [...overlay.colorDeclinations] }
      : {}),
    ...(overlay.isPinned ? { isPinned: true } : product.isPinned ? { isPinned: true } : {}),
    ...(overlay.displayOrder !== undefined
      ? { displayOrder: overlay.displayOrder }
      : product.displayOrder !== undefined
        ? { displayOrder: product.displayOrder }
        : {}),
    ...(overlay.adminCreatedAt
      ? { adminCreatedAt: overlay.adminCreatedAt }
      : product.adminCreatedAt
        ? { adminCreatedAt: product.adminCreatedAt }
        : {}),
  };
}

function normalizeRemovedCatalogIds(ids: readonly (string | number)[] | undefined): Set<number> {
  const removed = new Set<number>();
  if (!ids?.length) return removed;
  for (const raw of ids) {
    const id = Number(String(raw).trim());
    if (Number.isInteger(id) && id > 0) removed.add(id);
  }
  return removed;
}

/** Id masqué boutique : tombstone CRM ou verso dont le recto est retiré. */
export function isShopCatalogIdHidden(
  catalogId: number,
  removed: ReadonlySet<number>,
): boolean {
  if (removed.has(catalogId)) return true;
  const rectoId = resolveRectoCatalogId(catalogId);
  return rectoId !== catalogId && removed.has(rectoId);
}

export function overlayShopCatalog(
  fallback: ShopProduct[],
  live: CatalogProduct[],
  removedCatalogIds: readonly (string | number)[] = [],
): ShopProduct[] {
  const removed = normalizeRemovedCatalogIds(removedCatalogIds);
  const liveById = new Map(live.map((product) => [product.id, product]));

  const merged = fallback.map((product) => {
    const resolved = resolveLiveOverlay(product.id, liveById);
    const canonicalImage = CATALOG_CANONICAL_IMAGE_BY_ID[product.id];
    if (!resolved) {
      const image = product.image.trim() && !isDraftPlaceholderImage(product.image)
        ? product.image
        : (canonicalImage ?? product.image);
      const category = resolveShopCategory(product.id, product.category);
      const next =
        image !== product.image || category !== product.category
          ? { ...product, image, category }
          : product;
      return next;
    }
    return applyCatalogOverlay(product, resolved.overlay, resolved.inheritedFromRecto);
  });

  const extras = live
    .filter((product) => !fallback.some((item) => item.id === product.id))
    .map(catalogToShopProduct)
    .map((product) => {
      const liveImage = product.image.trim();
      if (liveImage && !isDraftPlaceholderImage(liveImage)) return product;
      const canonicalImage = CATALOG_CANONICAL_IMAGE_BY_ID[product.id];
      return canonicalImage ? { ...product, image: canonicalImage } : product;
    });

  return [...merged, ...extras].filter((product) => !isShopCatalogIdHidden(product.id, removed));
}

function sizesFromApiRecord(record: Record<string, unknown>): CatalogProduct["sizes"] {
  if (!Array.isArray(record.sizes)) return ["S", "M", "L", "XL"];
  const sizes = record.sizes
    .map((item) => (typeof item === "string" ? item.trim() : ""))
    .filter((item): item is string => item.length > 0 && isProductSize(item));
  return sizes.length > 0 ? (Array.from(new Set(sizes)) as CatalogProduct["sizes"]) : ["S", "M", "L", "XL"];
}

export function shopProductFromUnknown(value: unknown): CatalogProduct | null {
  if (!value || typeof value !== "object") return null;
  const record = value as Record<string, unknown>;
  const id = Number(record.id);
  const price = Number(record.price);
  if (!Number.isInteger(id) || id <= 0 || !Number.isFinite(price) || price <= 0) return null;
  if (typeof record.name !== "string" || typeof record.image !== "string") return null;
  const inStockFromApi = typeof record.inStock === "boolean" ? record.inStock : undefined;
  const stockFromApi: number | undefined =
    inStockFromApi !== undefined
      ? inStockFromApi
        ? 1
        : 0
      : record.stock !== undefined && Number.isFinite(Number(record.stock))
        ? Math.max(0, Number(record.stock))
        : undefined;
  const printPlacement = normalizePrintPlacement(record.printPlacement);
  return {
    id,
    sku: typeof record.sku === "string" ? record.sku : `DINGRY-${id}`,
    name: record.name,
    variant: typeof record.variant === "string" ? record.variant : "",
    ...(typeof record.imageBack === "string" && record.imageBack.trim()
      ? { imageBack: record.imageBack.trim() }
      : {}),
    description: typeof record.description === "string" ? record.description : "",
    fabrication: typeof record.fabrication === "string" ? record.fabrication : undefined,
    limitedEdition: typeof record.limitedEdition === "string" ? record.limitedEdition : undefined,
    materials: typeof record.materials === "string" ? record.materials : undefined,
    sizeGuide: typeof record.sizeGuide === "string" ? record.sizeGuide : undefined,
    ...(printPlacement ? { printPlacement } : {}),
    category: normalizeAdminCategory(typeof record.category === "string" ? record.category : "Tee"),
    price,
    image: record.image,
    sizes: sizesFromApiRecord(record),
    stock: stockFromApi,
  };
}

  return (
    category === "Tee" ||
    category === "T-shirts" ||
    category === SHOP_TEE_CATEGORY
  );
}

export function isShopCategory(category: string): category is ShopCategory {
  return (SHOP_CATEGORY_TABS as readonly string[]).includes(category) || isShopTeeCategory(category);
}

/** Admin → libellé boutique (onglets + filtres). */
export function resolveShopCategory(productId: number, category: string): ShopCategory | string {
  const slug = normalizeCatalogCategory(category);
  if (slug) return SHOP_CATEGORY_BY_MENU_SLUG[slug];
  if (DINGRY_SWEAT_PRODUCT_IDS.has(productId)) return "Sweats";
  if (DINGRY_TEE_PRODUCT_IDS.has(productId)) return SHOP_TEE_CATEGORY;
  if (DINGRY_JERSEY_PRODUCT_IDS.has(productId)) return "Jersey";
  if (category === "Tee" || category === "T-shirts") return SHOP_TEE_CATEGORY;
  return category;
}

export function normalizeAdminCategory(value: string): ProductCategory {
  return normalizeProductCategory(value);
}

/** Cartes grille : hors doublons retirés et hors vues verso. La catégorie ne filtre pas ici. */
export function toCatalogCards<T extends { id: number; variant?: string }>(products: readonly T[]): T[] {
  return products.filter(
    (product) => isActiveShopProduct(product) && !/\bVERSO\b/i.test(product.variant ?? ""),
  );
}

/** Grille : catégorie stricte, puis cartes recto. `all` garde aussi les catégories inconnues. */
export function filterCatalogProducts(
  products: ShopProduct[],
  tab: CatalogFilter,
): ShopProduct[] {
  return toCatalogCards(filterByCategory(products, tab));
}

/** Vue « Collection » : blocs produits dans l’ordre du menu (Hoodies → Caps). */
export function buildCatalogMenuSections(products: readonly ShopProduct[]): CatalogMenuSection[] {

  return CATALOG_CATEGORIES.map((category) => ({
    category,
    label: CATALOG_FILTER_LABEL[category],
    products: sortCatalogProducts(
      cards.filter((product) => normalizeCatalogCategory(product.category) === category),
      "featured",
    ),
  })).filter(keepCatalogMenuSection);
}

import type { AdminProduct, AdminStore } from "@/lib/admin/types";
import { toCatalogProduct } from "@/lib/admin/types";
import { overlayShopCatalog, type ShopProduct } from "@/lib/catalog";
import { PRODUCTS } from "@/lib/shop-data";

export type ShopCatalogSnapshot = {
  shopById: ReadonlyMap<number, ShopProduct>;
  removed: ReadonlySet<number>;
  adminById: ReadonlyMap<number, AdminProduct>;
};

export function removedCatalogIdSet(store: AdminStore): ReadonlySet<number> {
  return new Set(
    (store.removedCatalogIds ?? [])
      .map((id) => Number(id))
      .filter((catalogId) => Number.isInteger(catalogId) && catalogId > 0),
  );
}

export function publicLiveCatalogRows(
  store: AdminStore,
): NonNullable<ReturnType<typeof toCatalogProduct>>[] {
  return store.products
    .map((row) => toCatalogProduct(row))
    .filter((row): row is NonNullable<ReturnType<typeof toCatalogProduct>> => row !== null);
}

function adminProductById(store: AdminStore): ReadonlyMap<number, AdminProduct> {
  const map = new Map<number, AdminProduct>();
  for (const product of store.products) {
    const id = Number(product.id);
    if (Number.isInteger(id) && id > 0) map.set(id, product);
  }
  return map;
}

/** Même surface que la vitrine (`overlayShopCatalog` + tombstones). */
export function buildShopCatalogSnapshot(store: AdminStore): ShopCatalogSnapshot {
  const removed = removedCatalogIdSet(store);
  const live = publicLiveCatalogRows(store);
  const shopById = new Map(
    overlayShopCatalog(PRODUCTS, live, [...removed]).map((product) => [product.id, product]),
  );
  return { shopById, removed, adminById: adminProductById(store) };
}

import { NextResponse } from "next/server";
import { StoreUnavailableError } from "@/lib/admin/store";
import { getCachedCatalogProductRows } from "@/lib/server/shop-catalog-cache";
import { isInStockFromQuantity, readStockQuantities } from "@/lib/stock";

export const runtime = "nodejs";

export async function GET() {
  try {
    const { products: catalogRows, removedCatalogIds } = await getCachedCatalogProductRows();
    const stockById = await readStockQuantities(catalogRows.map((product) => product.id));
    const products = catalogRows.map((catalog) => {
      const live = stockById.get(catalog.id) ?? "missing";
      const inStock = isInStockFromQuantity(live);
      const { stock: _stock, ...publicProduct } = catalog;
      return { ...publicProduct, inStock };
    });
    return NextResponse.json(
      { products, removedCatalogIds },
      {
        headers: {
          "Cache-Control": "private, no-store, max-age=0, must-revalidate",
        },
      },
    );
  } catch (error) {
    if (error instanceof StoreUnavailableError) {
      console.error("Catalogue injoignable", error.name);
      return NextResponse.json({ error: "Catalogue momentanément indisponible." }, { status: 503 });
    }
    console.error("Catalogue injoignable", error instanceof Error ? error.name : "erreur");
    return NextResponse.json({ error: "Catalogue momentanément indisponible." }, { status: 503 });
  }
}

import { DeleteObjectCommand, PutObjectCommand, S3Client } from "@aws-sdk/client-s3";
import { compressImageToWebp } from "@/lib/supabase/compress-image";
import {
  isR2ImageStorageEnabled,
  isR2PublicUrl,
  objectKeyFromR2PublicUrl,
  r2BucketName,
  r2CredentialFormatError,
  r2PublicUrlForKey,
  r2S3Endpoint,
} from "@/lib/r2/config";

export { isR2ImageStorageEnabled } from "@/lib/r2/config";

export type R2StoredImage = {
  imageUrl: string;
  blobUrl: string | null;
  blobPathname: string | null;
};

let cachedClient: S3Client | null = null;

function r2Client(): S3Client {
  const accessKeyId = (process.env.R2_ACCESS_KEY_ID ?? "").trim();
  const secretAccessKey = (process.env.R2_SECRET_ACCESS_KEY ?? "").trim();
  if (!accessKeyId || !secretAccessKey) {
    throw new Error("R2 : R2_ACCESS_KEY_ID et R2_SECRET_ACCESS_KEY requis.");
  }
  const formatError = r2CredentialFormatError();
  if (formatError) {
    throw new Error(`R2 : ${formatError}`);
  }
  cachedClient ??= new S3Client({
    region: "auto",
    endpoint: r2S3Endpoint(),
    credentials: { accessKeyId, secretAccessKey },
  });
  return cachedClient;
}

export function r2ObjectKey(folder: string, fileName: string): string {
  const stamp = `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
  const safeFolder = folder.replace(/^\/+|\/+$/g, "");
  return `${safeFolder}/${stamp}-${fileName}`;
}

export async function uploadR2PublicImage(objectKey: string, file: File): Promise<R2StoredImage> {
  if (!isR2ImageStorageEnabled()) {
    throw new Error("Stockage R2 non configuré.");
  }
  const key = objectKey.replace(/^\/+/, "");
  const bytes = await compressImageToWebp(Buffer.from(await file.arrayBuffer()));
  await r2Client().send(
    new PutObjectCommand({
      Bucket: r2BucketName(),
      Key: key,
      Body: bytes,
      ContentType: "image/webp",
      CacheControl: "public, max-age=31536000, immutable",
    }),
  );
  const imageUrl = r2PublicUrlForKey(key);
  return { imageUrl, blobUrl: imageUrl, blobPathname: key };
}

export async function deleteR2Object(objectKey: string | null | undefined): Promise<void> {
  if (!objectKey?.trim() || !isR2ImageStorageEnabled()) return;
  const key = objectKey.replace(/^\/+/, "");
  try {
    await r2Client().send(
      new DeleteObjectCommand({
        Bucket: r2BucketName(),
        Key: key,
      }),
    );
  } catch (error) {
    console.error("[r2] delete", error instanceof Error ? error.message : "erreur");
  }
}

export async function deleteR2PublicImage(
  imageUrl: string,
  objectKey: string | null | undefined,
): Promise<void> {
  if (!isR2PublicUrl(imageUrl) && !objectKey) return;
  const key = objectKey?.trim() || objectKeyFromR2PublicUrl(imageUrl);
  if (key) await deleteR2Object(key);
}

/** Nom du bucket R2 Cloudflare (sans le préfixe chemin objet `dingry/…`). */
export const R2_DEFAULT_BUCKET_NAME = "dingrybuckets";

/** URL publique navigateur (custom domain ou `pub-xxx.r2.dev`). */
export const R2_DEFAULT_PUBLIC_BASE = "https://pub-5c13c59abc20442390a5eee5cb7034b0.r2.dev";

function trimmed(name: string): string {
  return (process.env[name] ?? "").trim();
}

export function r2PublicBaseUrl(): string {
  const fromEnv = trimmed("R2_PUBLIC_BASE_URL");
  if (fromEnv) {
    if (fromEnv.startsWith("http://") || fromEnv.startsWith("https://")) {
      return fromEnv.replace(/\/$/, "");
    }
    return `https://${fromEnv.replace(/\/$/, "")}`;
  }
  const host = (trimmed("DINGRY_IMG_HOSTS").split(",")[0] ?? "").trim();
  if (host && !host.includes("://")) return `https://${host.replace(/\/$/, "")}`;
  if (host.startsWith("http")) {
    try {
      return new URL(host).origin;
    } catch {
      return R2_DEFAULT_PUBLIC_BASE;
    }
  }
  return R2_DEFAULT_PUBLIC_BASE;
}

export function r2BucketName(): string {
  return trimmed("R2_BUCKET_NAME") || R2_DEFAULT_BUCKET_NAME;
}

export function r2AccountId(): string {
  return trimmed("R2_ACCOUNT_ID");
}

export function r2S3Endpoint(): string {
  const custom = trimmed("R2_S3_ENDPOINT");
  if (custom) return custom.replace(/\/$/, "");
  const account = r2AccountId();
  if (!account) return "";
  return `https://${account}.r2.cloudflarestorage.com`;
}

/** Longueurs des clés « S3 API » Cloudflare R2 (pas le token API global du compte). */
export const R2_S3_ACCESS_KEY_ID_LENGTH = 32;
export const R2_S3_SECRET_ACCESS_KEY_LENGTH = 64;

/**
 * Détecte une config R2 typique (clés inversées, secret dans ACCESS_KEY_ID, etc.).
 * Ne retourne jamais les valeurs des clés.
 */
export function r2CredentialFormatError(): string | null {
  const accessKeyId = trimmed("R2_ACCESS_KEY_ID");
  const secretAccessKey = trimmed("R2_SECRET_ACCESS_KEY");
  if (!accessKeyId || !secretAccessKey) return null;

  if (accessKeyId.startsWith("cfat_") || secretAccessKey.startsWith("cfat_")) {
    return "Token API Cloudflare (préfixe cfat_) : incompatible avec l’upload S3. Créer un token « R2 / S3 API » (Access Key ID 32 + Secret 64) dans R2 → Manage R2 API Tokens.";
  }

  if (
    accessKeyId.length === R2_S3_SECRET_ACCESS_KEY_LENGTH &&
    secretAccessKey.length === R2_S3_ACCESS_KEY_ID_LENGTH
  ) {
    return "R2_ACCESS_KEY_ID et R2_SECRET_ACCESS_KEY semblent inversés (32 vs 64 caractères).";
  }

  if (accessKeyId.length !== R2_S3_ACCESS_KEY_ID_LENGTH) {
    const looksLikePublicUrl =
      /^https?:\/\//i.test(accessKeyId) ||
      accessKeyId.includes(".r2.dev") ||
      accessKeyId.includes("r2.cloudflarestorage.com");
    const hint = looksLikePublicUrl
      ? "Vous avez collé l’URL publique du bucket : la mettre uniquement dans R2_PUBLIC_BASE_URL / DINGRY_IMG_HOSTS, pas dans R2_ACCESS_KEY_ID."
      : accessKeyId.length === R2_S3_SECRET_ACCESS_KEY_LENGTH || accessKeyId.length === 65
        ? "La valeur ressemble à la clé secrète : la mettre dans R2_SECRET_ACCESS_KEY, pas R2_ACCESS_KEY_ID."
        : "Cloudflare → R2 → Manage R2 API Tokens → Create API token (permissions Object Read & Write sur le bucket) : utiliser l’« Access Key ID » (32 car.), pas le token API global ni l’URL publique.";
    return `R2_ACCESS_KEY_ID invalide (${accessKeyId.length} car., attendu ${R2_S3_ACCESS_KEY_ID_LENGTH}). ${hint}`;
  }

  if (secretAccessKey.length !== R2_S3_SECRET_ACCESS_KEY_LENGTH) {
    return `R2_SECRET_ACCESS_KEY invalide (${secretAccessKey.length} car., attendu ${R2_S3_SECRET_ACCESS_KEY_LENGTH}).`;
  }

  return null;
}

/** Upload CRM → R2 (clés API S3, pas l’URL publique seule). */
export function isR2ImageStorageEnabled(): boolean {
  if (process.env.VITEST === "true") {
    return Boolean(
      trimmed("R2_ACCESS_KEY_ID") &&
        trimmed("R2_SECRET_ACCESS_KEY") &&
        r2AccountId() &&
        r2BucketName(),
    );
  }
  return Boolean(
    trimmed("R2_ACCESS_KEY_ID") &&
      trimmed("R2_SECRET_ACCESS_KEY") &&
      r2AccountId() &&
      r2BucketName() &&
      r2S3Endpoint(),
  );
}

export function r2PublicUrlForKey(objectKey: string): string {
  const base = r2PublicBaseUrl();
  const path = objectKey.replace(/^\/+/, "");
  return `${base}/${path}`;
}

export function objectKeyFromR2PublicUrl(url: string): string | null {
  const trimmedUrl = url.trim();
  if (!trimmedUrl) return null;
  try {
    const base = r2PublicBaseUrl();
    const parsed = new URL(trimmedUrl);
    const baseParsed = new URL(base);
    if (parsed.origin !== baseParsed.origin) {
      if (!parsed.hostname.endsWith(".r2.dev")) return null;
    }
    const prefix = baseParsed.pathname.replace(/\/$/, "");
    let pathname = parsed.pathname;
    if (prefix && pathname.startsWith(prefix)) {
      pathname = pathname.slice(prefix.length);
    }
    const key = pathname.replace(/^\/+/, "");
    return key || null;
  } catch {
    return null;
  }
}

export function isR2PublicUrl(url: string | null | undefined): boolean {
  if (!url?.trim()) return false;
  const trimmedUrl = url.trim();
  if (trimmedUrl.includes(".r2.dev")) return true;
  try {
    return new URL(trimmedUrl).origin === new URL(r2PublicBaseUrl()).origin;
  } catch {
    return false;
  }
}

"use client";

import { useCallback, useEffect, useState } from "react";
import { overlayShopCatalog, shopProductFromUnknown, type ShopProduct } from "@/lib/catalog";
import { PRODUCTS } from "@/lib/shop-data";

export type CatalogState = "loading" | "ready" | "error";

export function useShopCatalog() {
  const [products, setProducts] = useState<ShopProduct[]>(PRODUCTS);
  const [catalogState, setCatalogState] = useState<CatalogState>("loading");
  const [retryCount, setRetryCount] = useState(0);

  useEffect(() => {
    let cancelled = false;
    async function loadCatalog() {
      setCatalogState((state) => (state === "ready" ? state : "loading"));
      try {
        const response = await fetch("/api/catalog");
        if (!response.ok) {
          if (!cancelled) {
            setProducts(PRODUCTS);
            setCatalogState("ready");
          }
          return;
        }
        const data = (await response.json()) as {
          products?: unknown;
          removedCatalogIds?: unknown;
        };
        const live = Array.isArray(data.products)
          ? data.products
              .map(shopProductFromUnknown)
              .filter((item): item is NonNullable<typeof item> => item !== null)
          : [];
        const removedCatalogIds = Array.isArray(data.removedCatalogIds)
          ? data.removedCatalogIds.filter(
              (id): id is number => typeof id === "number" && Number.isInteger(id) && id > 0,
            )
          : [];
        if (!cancelled) {
          setProducts(overlayShopCatalog(PRODUCTS, live, removedCatalogIds));
          setCatalogState("ready");
        }
      } catch {
        if (!cancelled) {
          setProducts(PRODUCTS);
          setCatalogState("ready");
        }
      }
    }
    void loadCatalog();
    return () => {
      cancelled = true;
    };
  }, [retryCount]);

  const retry = useCallback(() => setRetryCount((value) => value + 1), []);

  return { products, catalogState, retry };
}

const STORE_LOCK_RELEASE_LUA = `
if redis.call("get", KEYS[1]) == ARGV[1] then
  return redis.call("del", KEYS[1])
end
return 0`;

async function acquireStoreLock(): Promise<string | null> {
  const client = getSharedRedis();
  if (!client) return null;
  const token = randomUUID();
  for (let attempt = 0; attempt < STORE_LOCK_WAIT_ATTEMPTS; attempt++) {
    try {
      const result = await client.set(STORE_LOCK_KEY, token, { nx: true, ex: STORE_LOCK_TTL_SECONDS });
      if (result === "OK") return token;
    } catch (error) {
      console.error("[store] lock Redis", error instanceof Error ? error.name : "erreur");
      return null;
    }
    await new Promise((resolve) => setTimeout(resolve, 40 + attempt * 20));
  }
  throw new StoreConflictError();
}

async function releaseStoreLock(token: string | null): Promise<void> {
  if (!token) return;
  const client = getSharedRedis();
  if (!client) return;
  try {
    await client.eval(STORE_LOCK_RELEASE_LUA, [STORE_LOCK_KEY], [token]);
  } catch (error) {
    console.error("[store] unlock Redis", error instanceof Error ? error.name : "erreur");
  }
}

  if (typeof fields.sku === "string") product.sku = fields.sku;
  if (typeof fields.name === "string") product.name = fields.name;
  if (typeof fields.variant === "string") product.variant = fields.variant;
  if (typeof fields.description === "string") product.description = fields.description;
  if (typeof fields.fabrication === "string") product.fabrication = fields.fabrication;
  if (typeof fields.limitedEdition === "string") product.limitedEdition = fields.limitedEdition;
  if (typeof fields.materials === "string") product.materials = fields.materials;
  if (typeof fields.sizeGuide === "string") product.sizeGuide = fields.sizeGuide;
  if (fields.printPlacement !== undefined) product.printPlacement = fields.printPlacement;
  if (fields.category) product.category = fields.category;
  if (typeof fields.price === "number") product.price = fields.price;
  if (fields.sizes) product.sizes = fields.sizes;
  if (typeof fields.stock === "number") {
    product.stock = Math.max(0, Math.floor(fields.stock));
  }
  if (fields.publishStatus) product.publishStatus = fields.publishStatus;
  if (fields.colorDeclinations !== undefined) product.colorDeclinations = fields.colorDeclinations;
  if (typeof fields.isPinned === "boolean") product.isPinned = fields.isPinned;
  if (typeof fields.displayOrder === "number") {
    product.displayOrder = normalizeDisplayOrder(fields.displayOrder);
  }
  if (uploadedRecto) {
    product.imageUrl = uploadedRecto.imageUrl;
    product.blobUrl = uploadedRecto.blobUrl;
    product.blobPathname = uploadedRecto.blobPathname;
  } else if (fields.clearImage && product.publishStatus !== "live") {
    product.imageUrl = DRAFT_PRODUCT_PLACEHOLDER_IMAGE;
    product.blobUrl = null;
    product.blobPathname = null;
  }
  if (uploadedVerso) {
    product.imageBackUrl = uploadedVerso.imageUrl;
    product.imageBackBlobUrl = uploadedVerso.blobUrl;
    product.imageBackBlobPathname = uploadedVerso.blobPathname;
  } else if (clearImageBack) {
    product.imageBackUrl = "";
    product.imageBackBlobUrl = null;
    product.imageBackBlobPathname = null;
  }
  if (product.publishStatus === "live") {
    requirePublishableProduct(product);
  }
  return product;
}

async function findProductRecord(id: string): Promise<AdminProduct | undefined> {
  if (isSupabaseStoreEnabled()) {
    return (await getSupabaseProduct(id)) ?? undefined;
}

function tombstoneCatalogPair(tombstone: Set<string>, id: string): void {
  tombstone.add(id);
  const numericId = Number(id);
  if (!Number.isInteger(numericId) || numericId <= 0) return;
  if (isVersoCatalogId(numericId)) {
    tombstone.add(String(resolveRectoCatalogId(numericId)));
    return;
  }
  for (const [verso, recto] of VERSO_TO_RECTO_CATALOG_ID) {
    if (recto === numericId) tombstone.add(String(verso));
  }
}

export async function removeProduct(id: string): Promise<boolean> {
  if (isSupabaseStoreEnabled()) {
    const existing = await getSupabaseProduct(id);
    if (!existing) return false;
    const ok = await deleteSupabaseProduct(id);
    if (!ok) return false;
    if (SEEDED_PRODUCTS.some((seed) => seed.id === id)) {
      const tombstone = new Set<string>();
      tombstoneCatalogPair(tombstone, id);
      await addRemovedCatalogIds([...tombstone]);
      for (const extra of tombstone) {
        if (extra !== id) await deleteSupabaseProduct(extra);
      }
    }
    await deleteProductImage(existing);
    return true;
  }

  const removed = await withStoreOptimistic(
    (store) => {
      const product = store.products.find((item) => item.id === id);
      if (!product) {
        return { store, result: null, skipWrite: true };
      }
      const copy = { ...product, sizes: [...product.sizes] };
      store.products = store.products.filter((item) => item.id !== id);
      if (SEEDED_PRODUCTS.some((seed) => seed.id === id)) {
        const tombstone = new Set(store.removedCatalogIds ?? []);
        tombstoneCatalogPair(tombstone, id);
        store.removedCatalogIds = [...tombstone];
        store.products = store.products.filter((item) => !tombstone.has(item.id));
      }
      return { store, result: copy };
    },
    { adminRelaxedFinalize: true },
  );

  if (!removed) return false;
  await deleteProductImage(removed);
  return true;
}

  return snapshot.store.orders.find((item) => item.id === id);
